  \tcb@shadow@comp@ts{#3}{#1}%
  \tcb@shadow@comp@arc%
  \begin{pgfscope}%
    \pgfsetfillopacity{#4}%
    \tcb@boundary@base{#1+#3}{#2+#3}{\tcb@width+#1-#3}{\tcb@height+#2-#3}{#5}{#6}{#7}{#8}%
    \pgfusepath{fill}%
  \end{pgfscope}}

\newcommand*{\CreateSmartLargeOperator}[2]{
% Adapted from https://tex.stackexchange.com/questions/61598/new-command-with-cases-conditionals-if-thens/61600
    % Plain operator (no customization)
    \csdef{LargeOperator@#1@}{\csdef{LargeOperator@#1@Symbol}{\csuse{#1}}}
    % Operator with limits above and below symbol
    \csdef{LargeOperator@#1@l}{\csdef{LargeOperator@#1@Symbol}{\csuse{#1}\limits}}
    % Operato with limits beside symbol
    \csdef{LargeOperator@#1@n}{\csdef{LargeOperator@#1@Symbol}{\csuse{#1}\nolimits}}
    % Inline style operator
    \csdef{LargeOperator@#1@i}{\csdef{LargeOperator@#1@Symbol}{\textstyle\csuse{#1}}}
    % Display style operator
    \csdef{LargeOperator@#1@d}{\csdef{LargeOperator@#1@Symbol}{\displaystyle\csuse{#1}}}
    % Inline style operator with limits above and below symbol
    \csdef{LargeOperator@#1@il}{\csdef{LargeOperator@#1@Symbol}{\textstyle\csuse{#1}\limits}}
    % Inline style operator with limits beside symbol
    \csdef{LargeOperator@#1@in}{\csdef{LargeOperator@#1@Symbol}{\textstyle\csuse{#1}\nolimits}}
    % Display style operator with limits above and below symbol
    \csdef{LargeOperator@#1@dl}{\csdef{LargeOperator@#1@Symbol}{\displaystyle\csuse{#1}\limits}}
    % Display style operator with limits beside symbol
    \csdef{LargeOperator@#1@dn}{\csdef{LargeOperator@#1@Symbol}{\displaystyle\csuse{#1}\nolimits}}

% NOTE: In the command below, ##1 denotes the operator. It is NOT to be used as an argument!
\def\LargeOperatorSpecs@i##1,##2,##3,##4,##5,##6,##7\@nil{
% If no arguments, operate over n from 1 to infinity
    \ifx$##2$\csuse{LargeOperator@##1@Symbol}_{n=1}^{\infty}\else
    % If one argument, operate over n from ##2 to infinity
        \ifx$##3$\csuse{LargeOperator@##1@Symbol}_{n=##2}^{\infty}\else
        % If two arguments, operate over n from ##2 to ##3
            \ifx$##4$\csuse{LargeOperator@##1@Symbol}_{n=##2}^{##3}\else
            % If three arguments, operate over ##2 from ##3 to ##4
                \ifx$##5$\csuse{LargeOperator@##1@Symbol}_{##2=##3}^{##4}\else
                % If four arguments, operate over ##2 and ##3 from ##4 to ##5
                    \ifx$##6$\csuse{LargeOperator@##1@Symbol}_{##2,##3=##4}^{##5}\else

% ---------------------------------------------------------------------------
% Speed patch for xsim (only applied to the xsim release dated 2022/02/12).
% xsim's bookkeeping of exercise properties is quadratic in the number of
% exercises: every time a property is set, it scans every recorded entry of
% that property, and its list updates use \seq_if_in / \seq_gremove_duplicates.
% With ~500 exercises this costs several minutes per run. The replacements
% below keep exactly the same behaviour but use hash (csname) look-ups.
% ---------------------------------------------------------------------------
\ExplSyntaxOn
\str_if_eq:VnT \c_xsim_date_tl {2022/02/12}
  {
    % Index: for property #3 and exercise #1-#2, the list entry currently stored.
    \cs_new:Npn \__ncgfix_xsim_index:nnn #1#2#3
      { g__ncgfix_xsim_index_ \tl_to_str:n {#3::#1-#2} _str }

    % Remove the (unique) list entry of property #3 belonging to exercise #1-#2.
    \cs_gset_protected:Npn \__xsim_remove_property_from_list:nnn #1#2#3
      {
        \str_if_exist:cT { \__ncgfix_xsim_index:nnn {#1} {#2} {#3} }
          {
            \xsim_remove_from_list:nv {#3}
              { \__ncgfix_xsim_index:nnn {#1} {#2} {#3} }
            \cs_undefine:c { \__ncgfix_xsim_index:nnn {#1} {#2} {#3} }
          }
      }
    \cs_generate_variant:Nn \xsim_remove_from_list:nn {nv}
    \str_new:N \l__ncgfix_xsim_str

    % Same as the original, plus recording the new entry in the index.
    \cs_gset_protected:Npn \__xsim_set_property:nnnn #1#2#3#4
      {
        \xsim_verbose:n
          {
            Setting~ property~ `#3'~ of~ exercise~ type~ `#1'~ id~ `#2'~ to~
            value~ `#4'.
          }
        \xsim_if_property_noupdate:nF {#3}
          {
            % If the list already holds exactly this entry there is nothing
            % to do (the original removed it and appended it again, which
            % only changes the order of entries, not their content).
            \str_set:Nn \l__ncgfix_xsim_str {#1-#2=={#4}}
            \str_if_exist:cTF { \__ncgfix_xsim_index:nnn {#1} {#2} {#3} }
              {
                \str_if_eq:eeF
                  { \use:c { \__ncgfix_xsim_index:nnn {#1} {#2} {#3} } }
                  { \l__ncgfix_xsim_str }
                  {
                    \__xsim_remove_property_from_list:nnn {#1} {#2} {#3}
                    \xsim_add_to_list:nn {#3} {#1-#2=={#4}}
                  }
              }
              {
                % not indexed yet: either the entry was read from the aux
                % file (then it is already in the list) or it is new
                \xsim_attribute_if_set:nnF {#3} {#1-#2=={#4}}
                  { \xsim_add_to_list:nn {#3} {#1-#2=={#4}} }
              }
            \str_gset_eq:cN { \__ncgfix_xsim_index:nnn {#1} {#2} {#3} }
              \l__ncgfix_xsim_str
          }
        \xsim_attribute_set:nnn {#1!#2} {#3} {#4}
      }

    % Membership is tracked by the attribute xsim already keeps for every
    % list entry, so no linear \seq_if_in search is needed.
    \cs_gset_protected:Npn \xsim_add_to_list:nn #1#2
      {
        \xsim_verbose:n { Adding~ `#2'~ to~ list~ `#1'. }
        \str_set:Nn \l__xsim_tmpa_str {#2}
        \xsim_attribute_if_set:nnT {#1} {#2}
          { \seq_gremove_all:cV {g__xsim_list_#1_seq} \l__xsim_tmpa_str }
        \xsim_attribute_set:nnn {#1} {#2} {}
        \seq_gput_right:cV {g__xsim_list_#1_seq} \l__xsim_tmpa_str
      }
    \cs_gset_protected:Npn \xsim_remove_from_list:nn #1#2
      {
        \xsim_verbose:n { Removing~ `#2'~ to~ list~ `#1'. }
        \str_set:Nn \l__xsim_tmpa_str {#2}
        \xsim_attribute_if_set:nnT {#1} {#2}
          {
            \xsim_attribute_unset:nn {#1} {#2}
            \seq_gremove_all:cV {g__xsim_list_#1_seq} \l__xsim_tmpa_str
          }
      }

    % Duplicate removal (keeps first occurrences, like \seq_gremove_duplicates:N).
    % A hash-based scan first checks whether there are any duplicates at all
    % (normally there are none); only then is the sequence rebuilt.
    \bool_new:N \g__ncgfix_xsim_dup_bool
    \cs_new_protected:Npn \__ncgfix_xsim_gremove_duplicates:c #1
      {
        \bool_gset_false:N \g__ncgfix_xsim_dup_bool
        \group_begin:
          \seq_map_inline:cn {#1}
            {
              \cs_if_exist:cTF { __ncgfix_xsim_seen_ \tl_to_str:n {##1} }
                {
                  \bool_gset_true:N \g__ncgfix_xsim_dup_bool
                  \seq_map_break:
                }
                { \cs_set:cpn { __ncgfix_xsim_seen_ \tl_to_str:n {##1} } { } }
            }
        \group_end:
        \bool_if:NT \g__ncgfix_xsim_dup_bool
          { \seq_gremove_duplicates:c {#1} }
      }
    \cs_generate_variant:Nn \__ncgfix_xsim_index:nnn {VVn}
    \cs_gset_protected:Npn \xsim_set_list:n #1
      {
        \seq_gset_eq:cc {g__xsim_list_recorded_#1_seq} {g__xsim_list_#1_seq}
        \__ncgfix_xsim_gremove_duplicates:c {g__xsim_list_recorded_#1_seq}
        % index the entries of property lists read from the aux file
        \xsim_if_property_exist:nT {#1}
          {
            \seq_map_inline:cn {g__xsim_list_#1_seq}
              {
                \__xsim_extract_property_list_entry:NNNwww
                  \l__xsim_tmpa_str \l__xsim_tmpb_str \l__xsim_tmpc_str ##1 \q_stop
                \str_gset:cn
                  { \__ncgfix_xsim_index:VVn \l__xsim_tmpa_str \l__xsim_tmpb_str {#1} }
                  {##1}
              }
          }
      }
    \cs_gset_protected:Npn \xsim_update_list:n #1
      {
        \xsim_verbose:n { Updating~ list~ `#1'. }
        \__ncgfix_xsim_gremove_duplicates:c {g__xsim_list_#1_seq}
        \__ncgfix_xsim_gremove_duplicates:c {g__xsim_list_recorded_#1_seq}
        \str_set:Nx \l__xsim_tmpa_str
          { \seq_use:cn {g__xsim_list_#1_seq} {} }
        \str_set:Nx \l__xsim_tmpb_str
          { \seq_use:cn {g__xsim_list_recorded_#1_seq} {} }
        \str_if_eq:NNF \l__xsim_tmpa_str \l__xsim_tmpb_str
          { \xsim_rerun: }
        \bool_if:NF \l__xsim_empty_lists_bool
          {
            \xsim_add_property_to_aux:nn {#1}
              { { \seq_use:cV {g__xsim_list_#1_seq} \l__xsim_split_aux_lists_str } }
          }
      }
  }
